cmake_minimum_required(VERSION 3.16)
project(smbdrop LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Gui Core)
add_executable(smbdrop WIN32 main.cpp runner.cpp runner.h)
target_link_libraries(smbdrop PRIVATE Qt6::Widgets Qt6::Gui Qt6::Core)

target_sources(smbdrop PRIVATE ../licensing/license.cpp ../licensing/license.h)
target_include_directories(smbdrop PRIVATE ../licensing)
